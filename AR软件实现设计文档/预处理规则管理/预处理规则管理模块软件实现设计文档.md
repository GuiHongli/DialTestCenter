# 预处理规则管理模块软件实现设计文档

## 1 需求重述

### 1.1 需求背景
预处理规则管理模块是DialTestCenter系统的核心配置管理功能，用于管理拨测脚本执行时与微调中心交互所需的预处理规则。该模块支持规则的批量上传、解析、存储和管理，为拨测系统提供灵活可配置的规则管理能力。

### 1.2 需求功能介绍
预处理规则管理模块主要提供以下能力：

**预处理规则管理功能：**
- 支持上传ZIP包：仅ADMIN和OPERATOR有权限，上传时支持选择业务类型
- 上传ZIP包中可以嵌套目录，文件夹中放置多个预处理规则.json文件
- 解析JSON文件，提取出多条预处理规则
- 支持ZIP包下载：仅ADMIN、OPERATOR有权限
- 支持预处理规则的分页查询展示：所有人均有查看权限
- 支持基于业务类型、大类、app、规则名称的多维度筛选功能
- 规则名称联合所属业务必须唯一，支持强制覆盖功能
- **覆盖更新功能**：当上传同名ZIP包或相同规则时，前端弹出确认对话框，用户选择覆盖后更新数据
- 上传规则动作需要记录到操作记录模块

**双Tab展示功能：**
- **Tab1 - ZIP包管理**：展示上传的预处理规则ZIP包，支持ZIP包上传、下载、删除
- **Tab2 - 规则管理**：展示从多个ZIP包中提取出来的所有规则，支持多维度筛选

**预处理规则关键信息：**
- 规则名称：从JSON文件中提取
- 所属业务：用户选择
- 大类：从JSON的'category'字段提取
- App：从JSON的'apps'或'custom_rules'中的key值提取
- 内容（text）：以App为key，获取其对应的value值
- 是否是用户自定义：根据提取位置判断

**规则名称提取规则：**
- 非用户自定义：提取'category'字段和'apps'中的key值，拼接作为名称
- 用户自定义：提取'custom_rules'字段中的key值和内层的key值，拼接作为名称

**内容提取规则：**
- 以App为key，获取其对应的完整value值作为规则内容
- 对于非自定义规则：从'apps'对象中，以app名称为key获取对应的value
- 对于自定义规则：从'custom_rules'对象中，以app名称为key获取对应的value

示例：预处理规则1-sdf_xdf.json文件，提取出来的规则信息为：
- 规则名称：LIVE_douyin，内容：{"text": "douyin", "app_type": "app_type", "app_egn_protos": ["DouYin_Live", "Bytecdn"], "associate_rules": [{"rule_type": "xx", "top": 10}]}
- 规则名称：LIVE_kuaishou，内容：{"text": "kuaishou", "app_type": "app_type"}
- 规则名称：custome_rule_ghltest1_douyin，内容：{"text": "douyin", "app_type": "app_type", "app_egn_protos": ["DouYin_Live", "Bytecdn"]}
- 规则名称：custome_rule_ghltest1_kuaishou，内容：{"text": "kuaishou", "app_type": "app_type"}
- 规则名称：custome_rule_ghltest2_douyin，内容：{"text": "douyin", "app_type": "app_type", "app_egn_protos": ["DouYin_Live", "Bytecdn"]}
- 规则名称：custome_rule_ghltest2_kuaishou，内容：{"text": "kuaishou", "app_type": "app_type"}

## 2 功能实现分析

public PreprocessRuleUploadResult uploadPreprocessRules(MultipartFile zipFile, String businessZh, String businessEn, String description, String operatorUsername) {
    // 1. 文件验证
    validateZipFile(zipFile);
    
    // 2. 业务类型验证
    validateBusinessType(businessZh, businessEn);
    
    // 3. 保存ZIP包到数据库
    PreprocessRulePackage package = saveZipPackage(zipFile, businessZh, businessEn, description);
    
    // 4. 解压ZIP文件
    List<JsonFileInfo> jsonFiles = extractJsonFilesFromZip(zipFile);
    
    // 5. 解析JSON文件并提取规则
    List<PreprocessRuleInfo> rules = new ArrayList<>();
    for (JsonFileInfo jsonFile : jsonFiles) {
        List<PreprocessRuleInfo> fileRules = parseJsonFile(jsonFile, businessZh, businessEn, package.getId());
        rules.addAll(fileRules);
    }
    
    // 6. 检查重复性并处理
    List<PreprocessRuleInfo> processedRules = handleDuplicateRules(rules, businessZh);
    
    // 7. 保存规则到数据库（关联ZIP包）
    List<PreprocessRule> savedRules = savePreprocessRules(processedRules);
    
    // 8. 记录操作日志
    operationLogUtil.logPreprocessRuleUpload(operatorUsername, savedRules.size(), businessZh);
    
    // 9. 构建返回结果
    return buildUploadResult(package, savedRules);
}

private PreprocessRulePackage saveZipPackage(MultipartFile zipFile, String businessZh, String businessEn, String description) {
    PreprocessRulePackage package = new PreprocessRulePackage();

    package.setFileContent(zipFile.getBytes());
    package.setFileSize(zipFile.getSize());
    package.setDescription(description);

public ResponseEntity<Resource> downloadMultiplePreprocessRulePackages(List<Long> packageIds, String operatorUsername) {
    // 1. 验证ZIP包ID列表
    if (packageIds == null || packageIds.isEmpty()) {
        throw new IllegalArgumentException("ZIP包ID列表不能为空");
    }
    
    // 2. 查询ZIP包信息
    List<PreprocessRulePackage> packages = preprocessRulePackageDao.findByIdIn(packageIds);
    if (packages.size() != packageIds.size()) {
        throw new IllegalArgumentException("部分ZIP包不存在");
    }
    
    // 3. 构建下载ZIP文件
    ByteArrayOutputStream zipOutputStream = new ByteArrayOutputStream();
    try (ZipOutputStream zipOut = new ZipOutputStream(zipOutputStream)) {
        
        // 按业务类型分组
        Map<String, List<PreprocessRulePackage>> packagesByBusiness = packages.stream()
            .collect(Collectors.groupingBy(PreprocessRulePackage::getBusinessZh));
        
        // 为每个业务类型创建目录
        for (Map.Entry<String, List<PreprocessRulePackage>> entry : packagesByBusiness.entrySet()) {
            String business = entry.getKey();
            List<PreprocessRulePackage> businessPackages = entry.getValue();
            
            // 创建业务目录
            String businessDir = business + "/";
            ZipEntry businessDirEntry = new ZipEntry(businessDir);
            zipOut.putNextEntry(businessDirEntry);
            zipOut.closeEntry();
            
            // 添加ZIP包文件
            for (PreprocessRulePackage package : businessPackages) {
                String fileName = businessDir + package.getPackageName();
                ZipEntry fileEntry = new ZipEntry(fileName);
                zipOut.putNextEntry(fileEntry);
                zipOut.write(package.getFileContent());
                zipOut.closeEntry();
            }
        }
    } catch (IOException e) {
        throw new RuntimeException("构建下载ZIP文件失败", e);
    }
    
    // 4. 记录操作日志
    operationLogUtil.logPreprocessRulePackagesDownload(operatorUsername, packageIds.size());
    
    // 5. 返回ZIP文件
    byte[] zipData = zipOutputStream.toByteArray();
    ByteArrayResource resource = new ByteArrayResource(zipData);
    
    return ResponseEntity.ok()
        .header(HttpHeaders.CONTENT_DISPOSITION, "attachment; filename=preprocess_rule_packages.zip")
        .contentType(MediaType.APPLICATION_OCTET_STREAM)
        .contentLength(zipData.length)
        .body(resource);
}

public List<PreprocessRuleInfo> parseJsonFile(JsonFileInfo jsonFile, String businessZh, String businessEn, Long packageId) {
    List<PreprocessRuleInfo> rules = new ArrayList<>();
    
    try {
        // 1. 解析JSON内容
        JsonNode rootNode = objectMapper.readTree(jsonFile.getContent());
        
        // 2. 提取category字段
        String category = rootNode.get("category").asText();
        
        // 3. 解析apps对象（非自定义规则）
        JsonNode appsNode = rootNode.get("apps");
        if (appsNode != null && appsNode.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> appsIterator = appsNode.fields();
            while (appsIterator.hasNext()) {
                Map.Entry<String, JsonNode> appEntry = appsIterator.next();
                String appName = appEntry.getKey();
                JsonNode appNode = appEntry.getValue();
                
                PreprocessRuleInfo rule = createNonCustomRule(category, appName, appNode, businessZh, businessEn, packageId);
                rules.add(rule);
            }
        }
        
        // 4. 解析custom_rules对象（自定义规则）
        JsonNode customRulesNode = rootNode.get("custom_rules");
        if (customRulesNode != null && customRulesNode.isObject()) {
            Iterator<Map.Entry<String, JsonNode>> customRulesIterator = customRulesNode.fields();
            while (customRulesIterator.hasNext()) {
                Map.Entry<String, JsonNode> customRuleEntry = customRulesIterator.next();
                String customRuleName = customRuleEntry.getKey();
                JsonNode customRuleNode = customRuleEntry.getValue();
                
                List<PreprocessRuleInfo> customRules = createCustomRules(customRuleName, customRuleNode, businessZh, businessEn, packageId);
                rules.addAll(customRules);
            }
        }
        
    } catch (Exception e) {
        throw new JsonParseException("解析JSON文件失败: " + jsonFile.getFileName(), e);
    }
    
    return rules;
}

private PreprocessRuleInfo createNonCustomRule(String category, String appName, JsonNode appNode, String businessZh, String businessEn, Long packageId) {
    PreprocessRuleInfo rule = new PreprocessRuleInfo();
    rule.setRuleName(category + "_" + appName);
    rule.setBusinessZh(businessZh);
    rule.setBusinessEn(businessEn);
    rule.setCategory(category);
    rule.setAppName(appName);
    // 以App为key，获取其对应的完整value值作为内容
    // appNode包含该app的完整配置信息，如text、app_type、app_egn_protos等
    rule.setContent(appNode.toString());
    rule.setIsCustom(false);
    rule.setPackageId(packageId);
    return rule;
}

private List<PreprocessRuleInfo> createCustomRules(String customRuleName, JsonNode customRuleNode, String businessZh, String businessEn, Long packageId) {
    List<PreprocessRuleInfo> rules = new ArrayList<>();
    
    if (customRuleNode.isObject()) {
        Iterator<Map.Entry<String, JsonNode>> appIterator = customRuleNode.fields();
        while (appIterator.hasNext()) {
            Map.Entry<String, JsonNode> appEntry = appIterator.next();
            String appName = appEntry.getKey();
            JsonNode appNode = appEntry.getValue();
            
            PreprocessRuleInfo rule = new PreprocessRuleInfo();
            rule.setRuleName(customRuleName + "_" + appName);
            rule.setBusinessZh(businessZh);
            rule.setBusinessEn(businessEn);
            rule.setCategory(customRuleName);
            rule.setAppName(appName);
            // 以App为key，获取其对应的完整value值作为内容
            // appNode包含该app在自定义规则中的完整配置信息
            rule.setContent(appNode.toString());
            rule.setIsCustom(true);
            rule.setPackageId(packageId);
            
            rules.add(rule);
        }
    }
    
    return rules;
}

### 2.6 功能点5：多维度筛选功能

#### 详细描述

### 2.8 功能点7：重复规则处理功能（覆盖更新）

#### 详细描述

**覆盖更新场景：**
1. **同名ZIP包覆盖**：当上传的ZIP包名称在同一业务类型下已存在时，提示用户是否覆盖
2. **同名规则覆盖**：当解析出的规则名称在同一业务类型下已存在时，自动覆盖（在覆盖模式下）

**前端实现流程：**
```
用户上传ZIP包
  ↓
后端检查同名ZIP包
  ↓
发现同名ZIP包 → 返回错误消息（包含"已存在同名ZIP包"）
  ↓
前端检测到重复错误
  ↓
弹出确认对话框
  ↓
用户选择"覆盖" → 重新上传（携带forceOverwrite=true标志）
用户选择"取消" → 取消上传
```

**后端实现流程：**
```
开始
  ↓
接收上传请求（包含forceOverwrite标志）
  ↓
检查同名ZIP包是否存在
  ↓
存在且forceOverwrite=false → 抛出异常"已存在同名ZIP包"
存在且forceOverwrite=true → 进入覆盖模式
不存在 → 进入新增模式
  ↓
【覆盖模式】
  ↓
更新ZIP包记录（file_content, file_size, description等）
  ↓
删除该包关联的所有旧规则（通过package_id）
  ↓
解析新ZIP包中的JSON文件
  ↓
对每条新规则：
  - 检查是否存在同名规则（rule_name + business_zh）
  - 存在则先删除同名规则
  - 插入新规则
  ↓
【新增模式】
  ↓
创建新ZIP包记录
  ↓
解析ZIP包中的JSON文件
  ↓
插入所有新规则
  ↓
记录操作日志
  ↓
返回成功结果
  ↓
结束
```

// Service层：实现覆盖逻辑
public String uploadPreprocessRulePackage(MultipartFile file, String businessZh, String businessEn, 
                                        String description, String operatorUsername, boolean forceOverwrite) throws IOException {
    
    // 1. 文件验证
    validateZipFile(file);
    
    String packageName = file.getOriginalFilename();
    
    // 2. 检查同名ZIP包
    PreprocessRulePackageEntity existingPackage = preprocessRulePackageDao.findByPackageNameAndBusiness(packageName, businessZh);
    
    if (existingPackage != null && !forceOverwrite) {
        // 已存在同名包且未强制覆盖，抛出异常
        throw new IllegalArgumentException("该业务类型下已存在同名ZIP包: " + packageName);
    }
    
    // 3. 读取文件内容
    byte[] fileContent = file.getBytes();
    
    PreprocessRulePackageEntity preprocessRulePackage;
    
    if (existingPackage != null && forceOverwrite) {
        // 【覆盖模式】更新现有记录
        logger.info("Overwriting existing package: {} (ID: {})", packageName, existingPackage.getId());
        
        existingPackage.setBusinessZh(businessZh);
        existingPackage.setBusinessEn(businessEn);
        existingPackage.setFileContent(fileContent);
        existingPackage.setFileSize(file.getSize());
        existingPackage.setDescription(description);
        
        preprocessRulePackageDao.update(existingPackage);
        preprocessRulePackage = existingPackage;
        
        // 删除该包关联的所有旧规则
        preprocessRuleDao.deleteByPackageId(existingPackage.getId());
        logger.info("Deleted existing rules for package ID: {}", existingPackage.getId());
        
    } else {
        // 【新增模式】创建新记录
        preprocessRulePackage = new PreprocessRulePackageEntity();
        preprocessRulePackage.setPackageName(packageName);
        preprocessRulePackage.setBusinessZh(businessZh);
        preprocessRulePackage.setBusinessEn(businessEn);
        preprocessRulePackage.setFileContent(fileContent);
        preprocessRulePackage.setFileSize(file.getSize());
        preprocessRulePackage.setDescription(description);
        
        preprocessRulePackageDao.insert(preprocessRulePackage);
    }
    
    // 4. 解析ZIP文件并提取规则
    int rulesCount = parseAndSaveRulesFromZip(fileContent, businessZh, businessEn, 
                                               preprocessRulePackage.getId(), forceOverwrite);
    
    // 5. 记录操作日志
    operationLogUtil.logPreprocessRulePackageUpload(operatorUsername, packageName, businessZh, businessEn);
    
    return buildSuccessResponse(preprocessRulePackage, rulesCount);
}

// 解析ZIP并保存规则（支持覆盖）
private int parseAndSaveRulesFromZip(byte[] zipFileContent, String businessZh, 
                                    String businessEn, Long packageId, boolean forceOverwrite) throws IOException {
    int rulesCount = 0;
    
    try (ByteArrayInputStream bais = new ByteArrayInputStream(zipFileContent);
         ZipInputStream zis = new ZipInputStream(bais)) {
        
        ZipEntry entry;
        while ((entry = zis.getNextEntry()) != null) {
            if (!entry.isDirectory() && entry.getName().toLowerCase().endsWith(".json")) {
                // 读取并解析JSON文件
                String jsonContent = readZipEntry(zis);
                List<PreprocessRule> rules = parseJsonFileContent(
                    jsonContent, businessZh, businessEn, packageId);
                
                // 保存规则到数据库
                for (PreprocessRule rule : rules) {
                    // 如果是覆盖模式，先删除同名规则
                    if (forceOverwrite) {
                        int deleted = preprocessRuleDao.deleteByRuleNameAndBusiness(
                            rule.getRuleName(), businessZh);
                        if (deleted > 0) {
                            logger.info("Deleted existing rule: {} for business: {}", 
                                       rule.getRuleName(), businessZh);
                        }
                    }
                    preprocessRuleDao.insert(rule);
                    rulesCount++;
                }
            }
            zis.closeEntry();
        }
    }
    
    return rulesCount;
}
